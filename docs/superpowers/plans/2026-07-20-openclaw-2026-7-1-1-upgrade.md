**Goal:** Upgrade new and existing MicroClaw installations to exactly OpenClaw `2026.7.1-1`, including protocol/plugin compatibility and local automatic rollback.

**Architecture:** Keep the current npm prefix and `~/.openclaw` state directory. A Python upgrade transaction snapshots the old npm package, command shims, and state before an in-place install; the Electron main process can recover an abandoned transaction before starting the Gateway. Compatibility changes are isolated into small helpers for version policy, protocol-v4 connect frames, v4 chat deltas, and hard Gateway restarts.

**Tech Stack:** Python 3.12 standard library, PowerShell 7, Node.js `>=22.22.3 <23`, TypeScript, Electron, Vue 3, Pinia, Vitest, npm, OpenClaw `2026.7.1-1`.

```python
# deployer/openclaw_version.py
import re

OPENCLAW_TARGET_VERSION = "2026.7.1-1"
NODE_FALLBACK_VERSION = "22.22.3"

_OPENCLAW_LIST_RE = re.compile(r"(?:^|\\s)openclaw@([^\\s]+)")


def _parse_node_version(value: str) -> tuple[int, int, int] | None:
    match = re.fullmatch(r"v?(\\d+)\\.(\\d+)\\.(\\d+)", value.strip())
    if not match:
        return None
    return tuple(int(part) for part in match.groups())


def is_supported_node_version(value: str) -> bool:
    version = _parse_node_version(value)
    if version is None:
        return False
    major, minor, patch = version
    if major == 22:
        return (minor, patch) >= (22, 3)
    if major == 24:
        return (minor, patch) >= (15, 0)
    if major >= 25:
        return version >= (25, 9, 0)
    return False


def extract_openclaw_version(npm_list_output: str) -> str | None:
    match = _OPENCLAW_LIST_RE.search(npm_list_output)
    return match.group(1) if match else None
```

```python
from __future__ import annotations

import json
import os
import shutil
import uuid
from dataclasses import asdict, dataclass, field
from datetime import UTC, datetime
from enum import StrEnum
from pathlib import Path


class UpgradePhase(StrEnum):
    BACKING_UP = "backing-up"
    INSTALLING = "installing"
    VERIFYING = "verifying"
    COMMITTED = "committed"
    ROLLING_BACK = "rolling-back"
    ROLLED_BACK = "rolled-back"
    ROLLBACK_FAILED = "rollback-failed"


ACTIVE_PHASES = {
    UpgradePhase.BACKING_UP,
    UpgradePhase.INSTALLING,
    UpgradePhase.VERIFYING,
    UpgradePhase.ROLLING_BACK,
}


@dataclass(frozen=True)
class OpenClawInstallation:
    version: str
    prefix: Path
    package_dir: Path
    entry_path: Path
    shim_paths: tuple[Path, ...]


@dataclass
class UpgradeManifest:
    schema_version: int
    transaction_id: str
    owner_pid: int
    source_version: str | None
    target_version: str
    prefix: str
    package_dir: str
    state_dir: str
    backup_dir: str
    shim_paths: list[str]
    package_existed: bool
    state_existed: bool
    phase: UpgradePhase
    created_at: str
    updated_at: str
    validation_results: dict[str, bool] = field(default_factory=dict)


def _now() -> str:
    return datetime.now(UTC).isoformat()


def _default_state_dir() -> Path:
    return Path.home() / ".openclaw"


def _atomic_json_write(path: Path, payload: dict) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temp = path.with_suffix(path.suffix + ".tmp")
    temp.write_text(json.dumps(payload, indent=2), encoding="utf-8")
    temp.replace(path)


def _require_within(path: Path, root: Path, label: str) -> Path:
    resolved = path.resolve(strict=False)
    allowed = root.resolve(strict=False)
    if resolved != allowed and not resolved.is_relative_to(allowed):
        raise ValueError(f"{label} escapes allowed root: {resolved}")
    return resolved
```

Add `OpenClawUpgradeTransaction` with these complete behaviors:

```python
class OpenClawUpgradeTransaction:
    def __init__(self, microclaw_root: Path, manifest: UpgradeManifest):
        self.microclaw_root = microclaw_root.resolve(strict=False)
        self.manifest = manifest
        self.manifest_path = self.microclaw_root / "upgrade" / "openclaw-upgrade.json"
        self.backup_root = self.microclaw_root / "backups" / "openclaw"
        self._validate_paths()

    @property
    def backup_dir(self) -> Path:
        return Path(self.manifest.backup_dir)

    @classmethod
    def create(
        cls,
        *,
        microclaw_root: Path,
        state_dir: Path,
        target_version: str,
        installation: OpenClawInstallation,
    ) -> OpenClawUpgradeTransaction:
        transaction_id = datetime.now(UTC).strftime("%Y%m%dT%H%M%SZ") + "-" + uuid.uuid4().hex[:8]
        backup_dir = microclaw_root / "backups" / "openclaw" / transaction_id
        timestamp = _now()
        manifest = UpgradeManifest(
            schema_version=1,
            transaction_id=transaction_id,
            owner_pid=os.getpid(),
            source_version=installation.version or None,
            target_version=target_version,
            prefix=str(installation.prefix.resolve(strict=False)),
            package_dir=str(installation.package_dir.resolve(strict=False)),
            state_dir=str(state_dir.resolve(strict=False)),
            backup_dir=str(backup_dir.resolve(strict=False)),
            shim_paths=[str(path.resolve(strict=False)) for path in installation.shim_paths],
            package_existed=installation.package_dir.exists(),
            state_existed=state_dir.exists(),
            phase=UpgradePhase.BACKING_UP,
            created_at=timestamp,
            updated_at=timestamp,
        )
        transaction = cls(microclaw_root, manifest)
        transaction._persist()
        return transaction

    @classmethod
    def load(cls, microclaw_root: Path) -> OpenClawUpgradeTransaction | None:
        manifest_path = microclaw_root / "upgrade" / "openclaw-upgrade.json"
        if not manifest_path.exists():
            return None
        data = json.loads(manifest_path.read_text(encoding="utf-8"))
        data["phase"] = UpgradePhase(data["phase"])
        return cls(microclaw_root, UpgradeManifest(**data))

    def _validate_paths(self) -> None:
        prefix = Path(self.manifest.prefix)
        _require_within(Path(self.manifest.package_dir), prefix, "package_dir")
        for shim in self.manifest.shim_paths:
            _require_within(Path(shim), prefix, "shim")
        expected_state = _default_state_dir()
        if Path(self.manifest.state_dir).resolve(strict=False) != expected_state.resolve(strict=False):
            raise ValueError("state_dir is not ~/.openclaw")
        _require_within(Path(self.manifest.backup_dir), self.backup_root, "backup_dir")

    def _persist(self) -> None:
        self.manifest.updated_at = _now()
        payload = asdict(self.manifest)
        payload["phase"] = self.manifest.phase.value
        _atomic_json_write(self.manifest_path, payload)
        if self.backup_dir.exists():
            _atomic_json_write(self.backup_dir / "transaction.json", payload)

    def set_phase(self, phase: UpgradePhase) -> None:
        self.manifest.phase = phase
        self._persist()

    @staticmethod
    def _ignore_state(directory: str, names: list[str]) -> set[str]:
        ignored = {name for name in names if name.endswith(".log")}
        current = Path(directory)
        if current.name == ".openclaw":
            ignored.update(name for name in names if name in {"compile-cache", "logs"})
        return ignored

    def backup(self) -> None:
        self.set_phase(UpgradePhase.BACKING_UP)
        self.backup_dir.mkdir(parents=True, exist_ok=False)
        package_dir = Path(self.manifest.package_dir)
        state_dir = Path(self.manifest.state_dir)
        if package_dir.exists():
            shutil.copytree(package_dir, self.backup_dir / "package")
        shims_dir = self.backup_dir / "shims"
        shims_dir.mkdir()
        for shim_value in self.manifest.shim_paths:
            shim = Path(shim_value)
            if shim.exists():
                shutil.copy2(shim, shims_dir / shim.name)
        if state_dir.exists():
            shutil.copytree(state_dir, self.backup_dir / "state", ignore=self._ignore_state)
        inventory = {
            str(path.relative_to(self.backup_dir)): path.stat().st_size
            for path in self.backup_dir.rglob("*")
            if path.is_file()
        }
        _atomic_json_write(self.backup_dir / "backup-files.json", inventory)
        self.set_phase(UpgradePhase.INSTALLING)

    def mark_verifying(self) -> None:
        self.set_phase(UpgradePhase.VERIFYING)

    def record_validation(self, name: str, passed: bool) -> None:
        self.manifest.validation_results[name] = passed
        self._persist()

    def commit(self) -> None:
        if not self.manifest.validation_results or not all(
            self.manifest.validation_results.values()
        ):
            raise RuntimeError("cannot commit before every validation passes")
        self.set_phase(UpgradePhase.COMMITTED)

    def rollback(self) -> None:
        original_phase = self.manifest.phase
        if original_phase == UpgradePhase.BACKING_UP:
            self.set_phase(UpgradePhase.ROLLED_BACK)
            return
        self.set_phase(UpgradePhase.ROLLING_BACK)
        package_dir = Path(self.manifest.package_dir)
        state_dir = Path(self.manifest.state_dir)
        failed_dir = self.backup_dir / "failed"
        failed_dir.mkdir(exist_ok=True)
        try:
            if package_dir.exists():
                shutil.rmtree(failed_dir / "package", ignore_errors=True)
                shutil.move(str(package_dir), failed_dir / "package")
            if self.manifest.package_existed:
                shutil.copytree(self.backup_dir / "package", package_dir)
            for shim_value in self.manifest.shim_paths:
                shim = Path(shim_value)
                shim.unlink(missing_ok=True)
                backup_shim = self.backup_dir / "shims" / shim.name
                if backup_shim.exists():
                    shim.parent.mkdir(parents=True, exist_ok=True)
                    shutil.copy2(backup_shim, shim)
            if state_dir.exists():
                shutil.rmtree(failed_dir / "state", ignore_errors=True)
                shutil.move(str(state_dir), failed_dir / "state")
            if self.manifest.state_existed:
                shutil.copytree(self.backup_dir / "state", state_dir)
            self.set_phase(UpgradePhase.ROLLED_BACK)
        except Exception:
            self.set_phase(UpgradePhase.ROLLBACK_FAILED)
            raise
```

```python
def process_is_alive(pid: int) -> bool:
    if pid <= 0:
        return False
    try:
        os.kill(pid, 0)
        return True
    except OSError:
        return False


def prune_previous_committed_backups(backup_root: Path, keep: Path) -> None:
    for candidate in backup_root.iterdir() if backup_root.exists() else ():
        if candidate.resolve(strict=False) == keep.resolve(strict=False) or not candidate.is_dir():
            continue
        manifest_path = candidate / "transaction.json"

            continue
        phase = json.loads(manifest_path.read_text(encoding="utf-8")).get("phase")
        if phase == UpgradePhase.COMMITTED.value:
            shutil.rmtree(candidate)
```

```python
        ("version", self._validate_installed_version),
        ("health", self._validate_gateway_health),

        for name, check in checks:

            if transaction is not None:

            if not passed:
                raise RuntimeError(f"OpenClaw validation failed: {name}")

    except Exception as exc:

        return False
    finally:

    transaction = getattr(self, "_openclaw_transaction", None)
    if transaction is None:
        return True
    transaction.commit()
    prune_previous_committed_backups(
        transaction.backup_root, keep=transaction.backup_dir
    )
    return True


def rollback_openclaw_upgrade(self) -> bool:
    transaction = getattr(self, "_openclaw_transaction", None)
    if transaction is None:
        return True
    try:
        transaction.rollback()
        if transaction.manifest.source_version is None:
            return True
        process = self._start_validation_gateway(
            expected_version=transaction.manifest.source_version

        try:
            return self._validate_gateway_health()
        finally:
            self._stop_validation_gateway(process)

            f"OpenClaw rollback failed; backup retained at {transaction.backup_dir}: {exc}"
```

```python
def _start_validation_gateway(
    self, expected_version: str | None = OPENCLAW_TARGET_VERSION
) -> subprocess.Popen:
    installation = self._detect_openclaw_installation()
    if installation is None or (
        expected_version is not None and installation.version != expected_version
    ):
        raise RuntimeError(f"expected OpenClaw package is not installed: {expected_version}")
    node = self.node_dir / "node.exe"
    if not node.exists() and self._node_bin:
        node = Path(self._node_bin) / "node.exe"
```

8. move failed package/state to `<backup_dir>/failed`, replacing an older
   diagnostic copy so retrying a `rolling-back` phase is idempotent;
9. copy package, shims, and state from backup;
10. atomically rewrite the phase to `rolled-back`;
11. atomically set `rollback-failed` and rethrow on failure.

write-temp/rename helper. Do not invoke npm or access the network.

Use this complete recovery flow:

```typescript
// desktop/src/openclaw-upgrade-recovery.ts
import fs from "fs";
import path from "path";

type UpgradePhase =
  | "backing-up"
  | "installing"
  | "verifying"
  | "committed"
  | "rolling-back"
  | "rolled-back"
  | "rollback-failed";

type UpgradeManifest = {
  schema_version: number;
  transaction_id: string;
  owner_pid: number;
  source_version: string | null;
  target_version: string;
  prefix: string;
  package_dir: string;
  state_dir: string;
  backup_dir: string;
  shim_paths: string[];
  package_existed: boolean;
  state_existed: boolean;
  phase: UpgradePhase;
  created_at: string;
  updated_at: string;
  validation_results: Record<string, boolean>;
};

export type UpgradeRecoveryResult =
  | { status: "none" | "committed" | "rolled-back" }
  | { status: "upgrade-in-progress"; ownerPid: number };

function isWithin(candidate: string, root: string): boolean {
  const relative = path.relative(path.resolve(root), path.resolve(candidate));
  return relative === "" || (!relative.startsWith("..") && !path.isAbsolute(relative));
}

function atomicWrite(pathname: string, manifest: UpgradeManifest): void {
  const temporary = `${pathname}.tmp`;
  manifest.updated_at = new Date().toISOString();
  fs.writeFileSync(temporary, JSON.stringify(manifest, null, 2), "utf-8");
  fs.renameSync(temporary, pathname);
}

function replaceWithBackup(current: string, backup: string, failed: string): void {
  fs.rmSync(failed, { recursive: true, force: true });
  if (fs.existsSync(current)) {
    fs.mkdirSync(path.dirname(failed), { recursive: true });
    fs.renameSync(current, failed);
  }
  if (fs.existsSync(backup)) {
    fs.mkdirSync(path.dirname(current), { recursive: true });
    fs.cpSync(backup, current, { recursive: true });
  }
}

function defaultProcessIsAlive(pid: number): boolean {
  if (!Number.isInteger(pid) || pid <= 0) return false;
  try {
    process.kill(pid, 0);
    return true;
  } catch {
    return false;
  }
}

export function recoverInterruptedOpenClawUpgrade(
  microclawRoot: string,
  isProcessAlive = defaultProcessIsAlive,
  expectedStateDir = path.join(path.dirname(path.resolve(microclawRoot)), ".openclaw"),
): UpgradeRecoveryResult {
  const manifestPath = path.join(microclawRoot, "upgrade", "openclaw-upgrade.json");
  if (!fs.existsSync(manifestPath)) return { status: "none" };
  const manifest = JSON.parse(fs.readFileSync(manifestPath, "utf-8")) as UpgradeManifest;
  if (manifest.schema_version !== 1) throw new Error("unsupported upgrade manifest schema");
  if (manifest.phase === "committed") return { status: "committed" };
  if (manifest.phase === "rolled-back") return { status: "rolled-back" };
  if (manifest.phase === "rollback-failed") {
    throw new Error(`previous rollback failed; backup retained at ${manifest.backup_dir}`);
  }
  if (isProcessAlive(manifest.owner_pid)) {
    return { status: "upgrade-in-progress", ownerPid: manifest.owner_pid };
  }

  const prefix = path.resolve(manifest.prefix);
  if (!isWithin(manifest.package_dir, prefix)) {
    throw new Error("package_dir is outside recorded prefix");
  }
  for (const shim of manifest.shim_paths) {
    if (!isWithin(shim, prefix)) throw new Error("shim is outside recorded prefix");
  }
  if (path.resolve(manifest.state_dir) !== path.resolve(expectedStateDir)) {
    throw new Error("state_dir is not the expected ~/.openclaw directory");
  }
  const backupRoot = path.join(microclawRoot, "backups", "openclaw");
  if (!isWithin(manifest.backup_dir, backupRoot)) {
    throw new Error("backup_dir is outside MicroClaw backup root");
  }

  if (manifest.phase === "backing-up") {
    manifest.phase = "rolled-back";
    atomicWrite(manifestPath, manifest);
    return { status: "rolled-back" };
  }

  manifest.phase = "rolling-back";
  atomicWrite(manifestPath, manifest);
  const failedRoot = path.join(manifest.backup_dir, "failed");
  try {
    replaceWithBackup(
      manifest.package_dir,
      manifest.package_existed ? path.join(manifest.backup_dir, "package") : "",
      path.join(failedRoot, "package"),
    );
    for (const shim of manifest.shim_paths) {
      fs.rmSync(shim, { force: true });
      const backupShim = path.join(manifest.backup_dir, "shims", path.basename(shim));
      if (fs.existsSync(backupShim)) {
        fs.mkdirSync(path.dirname(shim), { recursive: true });
        fs.copyFileSync(backupShim, shim);
      }
    }
    replaceWithBackup(
      manifest.state_dir,
      manifest.state_existed ? path.join(manifest.backup_dir, "state") : "",
      path.join(failedRoot, "state"),
    );
    manifest.phase = "rolled-back";
    atomicWrite(manifestPath, manifest);
    return { status: "rolled-back" };
  } catch (error) {
    manifest.phase = "rollback-failed";
    atomicWrite(manifestPath, manifest);
    throw error;
  }
}
```

```typescript
// desktop/src/gateway-lifecycle.ts
export type GatewayRestartDependencies = {
  stopClient: () => void;
  stopProcess: () => void;
  isPortOccupied: () => Promise<boolean>;
  startGateway: () => Promise<void>;
  sleep: (ms: number) => Promise<void>;
  timeoutMs: number;
  pollMs: number;
};

export async function hardRestartGateway(deps: GatewayRestartDependencies): Promise<void> {
  deps.stopClient();
  deps.stopProcess();
  const attempts = Math.max(1, Math.ceil(deps.timeoutMs / deps.pollMs));
  for (let attempt = 0; attempt < attempts; attempt += 1) {
    if (!(await deps.isPortOccupied())) {
      await deps.startGateway();
      return;
    }
    await deps.sleep(deps.pollMs);
  }
  throw new Error("gateway port remained occupied after process stop");
}
```
